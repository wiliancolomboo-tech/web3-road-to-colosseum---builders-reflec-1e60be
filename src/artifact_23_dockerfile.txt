FROM node:20-alpine AS builder
WORKDIR /app
COPY package*.json tsconfig.json ./
RUN npm ci
COPY src/ ./src
RUN npm run build

FROM node:20-alpine AS runner
WORKDIR /app
ENV NODE_ENV=production
COPY package*.json ./
RUN npm ci --only=production
COPY --from=builder /app/dist ./dist

# Create healthcheck endpoint script
RUN echo 'const http = require("http"); const server = http.createServer((_, res) => { res.writeHead(200); res.end("OK"); }); server.listen(8080);' > healthcheck.js

EXPOSE 8080
CMD ["node", "-e", "require('./healthcheck.js'); require('./dist/daemon.js');"]
